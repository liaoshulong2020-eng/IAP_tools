### 2.2 命令发送方法分类

#### 2.2.1 简单命令发送
```csharp
private void SendSimpleCommand(byte command)
{
    byte[] dataBytes = new byte[8] { 0x00, command, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00 };
    SendCANFrame(dataBytes, 2);
}
```

#### 2.2.2 16位数据命令（不带CRC）
```csharp
private void Send16BitSimple(byte command, ushort value)
{
    byte[] dataBytes = new byte[8]
    {
        0x00,                         // [0] 保留位
        command,                      // [1] 命令
        (byte)(value & 0xFF),         // [2] 低字节
        (byte)((value >> 8) & 0xFF),  // [3] 高字节
        0x00, 0x00, 0x00, 0x00       // [4-7] 保留位
    };
    SendCANFrame(dataBytes, 4);
}
```

#### 2.2.3 16位数据命令（带CRC）
```csharp
private void Send16BitWithCRC(byte command, ushort value)
{
    byte lowByte = (byte)(value & 0xFF);
    byte highByte = (byte)((value >> 8) & 0xFF);
    
    // 计算CRC
    byte[] crcData = new byte[3] { command, lowByte, highByte };
    byte crc = CalculateCRC8(crcData, 0, crcData.Length);
    
    byte[] dataBytes = new byte[8]
    {
        0x00, command, lowByte, highByte, crc, 0x00, 0x00, 0x00
    };
    SendCANFrame(dataBytes, 5);
}
```

#### 2.2.4 32位数据命令（带CRC）
```csharp
private void Send32BitWithCRC(byte command, uint value)
{
    byte byte0 = (byte)(value & 0xFF);
    byte byte1 = (byte)((value >> 8) & 0xFF);
    byte byte2 = (byte)((value >> 16) & 0xFF);
    byte byte3 = (byte)((value >> 24) & 0xFF);
    
    // 计算CRC
    byte[] crcData = new byte[5] { command, byte0, byte1, byte2, byte3 };
    byte crc = CalculateCRC8(crcData, 0, crcData.Length);
    
    byte[] dataBytes = new byte[8]
    {
        0x00, command, byte0, byte1, byte2, byte3, crc, 0x00
    };
    SendCANFrame(dataBytes, 7);
}
```

private void Command_NewFunction_Response(uint id, byte[] frameData, uint ch)

```csharp
// 在ParseAndDisplayData的switch中添加
case 0xE0:  // LED状态响应

    break;

// 实现响应处理方法
private void Command_LED_Response(uint id, byte[] frameData, uint ch)
{
    if (frameData == null || frameData.Length < 8) return;
    
    byte ledStatus = frameData[2];
    bool isLedOn = (ledStatus & 0x01) != 0;
    
    // 更新UI状态
    this.BeginInvoke(new Action(() =>
    {
        checkBox_LED.Checked = isLedOn;
    }));
    
    Console.WriteLine($"LED状态更新 - 通道{ch}: {(isLedOn ? "开启" : "关闭")}");
}
```

### 6.2 测试流程
1. **单元测试**: 先测试命令发送方法
2. **集成测试**: 测试UI交互和数据显示
3. **通信测试**: 使用CAN分析仪验证帧格式
4. **压力测试**: 测试高频率命令发送

### 6.3 常见问题排查
1. **命令无响应**: 检查CAN ID配置和波特率设置
2. **数据解析错误**: 验证字节序和数据格式
3. **UI更新异常**: 确保使用BeginInvoke进行跨线程调用
4. **CRC校验失败**: 检查CRC计算算法和数据范围
